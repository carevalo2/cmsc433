\sheettitle{02}{Functor, Applicative, and State types}{Topic mock}

\mockrules
Give the most general type or ``ill-typed'' with a reason. All parts are
independent. Keep function-valued payloads and necessary constraints.
\begin{enumerate}[label=(\alph*)]
\item \code{get >>= put}
\answer{[Answer.]}
\item \code{modify tail}
\answer{[Answer.]}
\item \code{put 'x' >> put True}
\answer{[Answer.]}
\item \code{(:) <$> get}
\answer{[Stored type; returned type; whole type.]}
\item \code{\f mx -> f <$> mx}
\answer{[Most general type.]}
\item \code{get >>= \x -> return (x, x + 2)}
\answer{[Most general type and constraint.]}
\end{enumerate}
\textbf{Supplied signatures}
\par\noindent\begin{minipage}{\linewidth}
\begin{lstlisting}
get :: State s s
put :: s -> State s ()
modify :: (s -> s) -> State s ()
tail :: [a] -> [a]
(:) :: a -> [a] -> [a]
(<$>) :: Functor f => (a -> b) -> f a -> f b
(>>=) :: Monad m => m a -> (a -> m b) -> m b
(>>) :: Monad m => m a -> m b -> m b
return :: Monad m => a -> m a
\end{lstlisting}
\end{minipage}\par
